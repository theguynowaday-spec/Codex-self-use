```js
function requestJson(options, onSuccess, onError) {
    var xhr = new XMLHttpRequest();
    var method = options.method || "GET";
    var url = options.url;
    var body = options.body || null;
    var headers = options.headers || {};

    xhr.open(method, url, true);

    for (var h in headers) {
        if (headers.hasOwnProperty(h)) {
            xhr.setRequestHeader(h, headers[h]);
        }
    }

    xhr.onreadystatechange = function () {
        if (xhr.readyState !== 4) {
            return;
        }

        if (xhr.status >= 200 && xhr.status < 300) {
            try {
                onSuccess(JSON.parse(xhr.responseText), xhr);
            } catch (e) {
                if (onError) onError(e, xhr);
            }
        } else {
            if (onError) onError(new Error("HTTP " + xhr.status), xhr);
        }
    };

    xhr.onerror = function () {
        if (onError) onError(new Error("Network error"), xhr);
    };

    xhr.send(body);
}

Что важно:

headers задаются через цикл;
JSON парсится внутри try/catch;

function toQueryString(params) {
    var parts = [];
    var key;

    for (key in params) {
        if (params.hasOwnProperty(key) && params[key] !== undefined && params[key] !== null) {
            parts.push(encodeURIComponent(key) + "=" + encodeURIComponent(params[key]));

    return parts.join("&");
}

ДОСТАЁМ РЕВИЗИЮ И WIKITEXT
function extractPageContent(data) {
    var pages = data.query.pages;
    var pageId;
    var page;
    var revs;
    var text = "";

    for (pageId in pages) {
        if (pages.hasOwnProperty(pageId)) {
            page = pages[pageId];

            if (page.revisions && page.revisions.length) {
                revs = page.revisions[0];
                text = revs["*"] || revs.slots && revs.slots.main && revs.slots.main["*"] || "";
                return {
                    title: page.title,
                    pageid: page.pageid,
                    revid: revs.revid,
                    timestamp: revs.timestamp,
                    text: text
                };
            }
        }
    }

    return null;
}

ПЕРЕДЕЛЫВАЕМ WIKITEXT В ПРОСТЫЙ ТЕКСТ
function stripWikiMarkup(text) {
    if (!text) return "";

    text = text.replace(/\{\{[^{}]*\}\}/g, "");
    text = text.replace(/\[\[(?:[^|\]]*\|)?([^\]]+)\]\]/g, "$1");
    text = text.replace(/'''/g, "");
    text = text.replace(/''/g, "");
    text = text.replace(/<!--[\s\S]*?-->/g, "");
    text = text.replace(/^\s+|\s+$/g, "");

    return text;
}

function getCategoryMembers(baseApi, categoryTitle, onItem, onDone, onError) {
    var cmcontinue = null;

    function step() {
        var params = {
            action: "query",
            list: "categorymembers",
            cmtitle: categoryTitle,
            cmlimit: 50,
            format: "json"
        };

        if (cmcontinue) {
            params.cmcontinue = cmcontinue;
        }

        var url = buildApiUrl(baseApi, params);

        apiGetJson(url, function (data) {
            var members = data.query && data.query.categorymembers || [];
            var i;

            for (i = 0; i < members.length; i++) {
                onItem(members[i]);
            }

            if (data.continue && data.continue.cmcontinue) {
                cmcontinue = data.continue.cmcontinue;
                step();
            } else {
                if (onDone) onDone();
            }
        }, onError);
    }

    step();
}

function safeEdit(baseApi, title, fetchContentFn, transformFn, summary, onDone, onError) {
    fetchContentFn(title, function (page) {
        var currentText = page.text || "";
        var nextText = transformFn(currentText);

        if (currentText === nextText) {
            if (onDone) onDone({ skipped: true, reason: "no changes" });
            return;
        }

        getCsrfToken(baseApi, function (token) {
            editPage(baseApi, title, nextText, token, summary, function (result) {
                if (onDone) onDone(result);
            }, onError);
        }, onError);
    }, onError);
}

function replaceOnce(text, from, to) {
    var idx = text.indexOf(from);
    if (idx === -1) return text;
    return text.slice(0, idx) + to + text.slice(idx + from.length);
}

РЕАЛИЗАЦИЯ RETRY С ПАУЗОЙ
function retryAsync(taskFn, attempts, delay, onSuccess, onError) {
    function run(remaining) {
        taskFn(function success(result) {
            onSuccess(result);
        }, function fail(err) {
            if (remaining > 1) {
                setTimeout(function () {
                    run(remaining - 1);
                }, delay);
            } else {
                onError(err);
            }
        });
    }

    run(attempts);
}

function runQueue(tasks, done) {
    var i = 0;

    function next() {
        if (i >= tasks.length) {
            if (done) done();
            return;
        }

        tasks[i++](function () {
            next();
        });
    }

    next();
}

(function () {
    "use strict";

    var BASE_API = "https://example.fandom.com/api.php";
    var PAGE_TITLE = "Sandbox";
    var SUMMARY = "Automated cleanup via ES5 bot";
    var WAIT_MS = 2000;

    function requestJson(options, onSuccess, onError) {
        var xhr = new XMLHttpRequest();
        xhr.open(options.method || "GET", options.url, true);

        if (options.headers) {
            for (var h in options.headers) {
                if (options.headers.hasOwnProperty(h)) {
                    xhr.setRequestHeader(h, options.headers[h]);
                }
            }
        }

        xhr.onreadystatechange = function () {
            if (xhr.readyState !== 4) return;

            if (xhr.status >= 200 && xhr.status < 300) {
                try {
                    onSuccess(JSON.parse(xhr.responseText), xhr);
                } catch (e) {
                    if (onError) onError(e, xhr);
                }
            } else {
                if (onError) onError(new Error("HTTP " + xhr.status), xhr);
            }
        };

        xhr.onerror = function () {
            if (onError) onError(new Error("Network error"), xhr);
        };

        xhr.send(options.body || null);
    }

    function toQueryString(params) {
        var parts = [];
        var key;

        for (key in params) {
            if (params.hasOwnProperty(key) && params[key] !== undefined && params[key] !== null) {
                parts.push(encodeURIComponent(key) + "=" + encodeURIComponent(params[key]));
            }
        }

        return parts.join("&");
    }

    function apiGet(params, onSuccess, onError) {
        var url = BASE_API + "?" + toQueryString(params);
        requestJson({
            method: "GET",
            url: url
        }, onSuccess, onError);
    }

    function apiPost(body, onSuccess, onError) {
        requestJson({
            method: "POST",
            url: BASE_API,
            body: body,
            headers: {
                "Content-Type": "application/x-www-form-urlencoded; charset=UTF-8"
            }
        }, onSuccess, onError);
    }

    function getCsrfToken(onSuccess, onError) {
        apiGet({
            action: "query",
            meta: "tokens",
            type: "csrf",
            format: "json"
        }, function (data) {
            var token = data && data.query && data.query.tokens && data.query.tokens.csrftoken;
            if (!token) {
                onError(new Error("CSRF token not found"));
                return;
            }
            onSuccess(token);
        }, onError);
    }

    function readPage(title, onSuccess, onError) {
        apiGet({
            action: "query",
            prop: "revisions",
            titles: title,
            rvprop: "content|timestamp|ids",
            format: "json"
        }, function (data) {
            var pages = data && data.query && data.query.pages;
            var id;
            var page;
            var rev;
            var text;

            if (!pages) {
                onError(new Error("Pages object missing"));
                return;
            }

            for (id in pages) {
                if (pages.hasOwnProperty(id)) {
                    page = pages[id];
                    if (page.missing) {
                        onError(new Error("Page missing: " + title));
                        return;
                    }
                    rev = page.revisions && page.revisions[0];
                    if (!rev) {
                        onError(new Error("No revisions found"));
                        return;
                    }
                    text = rev["*"] || (rev.slots && rev.slots.main && rev.slots.main["*"]) || "";
                    onSuccess({
                        title: page.title,
                        pageid: page.pageid,
                        revid: rev.revid,
                        timestamp: rev.timestamp,
                        text: text
                    });
                    return;
                }
            }

            onError(new Error("Page not found in response"));
        }, onError);
    }

    function transformText(text) {
        var out = text;

        out = out.replace(/\s+\n/g, "\n");
        out = out.replace(/\n{3,}/g, "\n\n");

        // Example cleanup:
        // remove double spaces
        out = out.replace(/[ \t]{2,}/g, " ");

        return out;
    }

    function editPage(title, newText, token, baseTimestamp, onSuccess, onError) {
        var body = toQueryString({
            action: "edit",
            title: title,
            text: newText,
            summary: SUMMARY,
            token: token,
            basetimestamp: baseTimestamp,
            starttimestamp: (new Date()).toISOString(),
            format: "json"
        });

        apiPost(body, function (data) {
            if (data && data.edit && data.edit.result === "Success") {
                onSuccess(data);
                return;
            }

            if (data && data.error) {

    }

                        console.log("Page updated:", page.title);
                    }, function (err) {

                    });
                }, WAIT_MS);
            }, function (err) {
                console.log("Token error:", err.message);
            });

function processPages(pages, handler, done) {
    var i = 0;

    function step() {
        if (i >= pages.length) {
            if (done) done();
            return;
        }

        handler(pages[i], function () {
            i++;
            setTimeout(step, 1000);
        });
    }

    step();
}

function getOrDefault(obj, path, fallback) {
    var parts = path.split(".");
    var cur = obj;
    var i;
    var part;

    for (i = 0; i < parts.length; i++) {
        part = parts[i];
        if (!cur || typeof cur !== "object" || !(part in cur)) {
            return fallback;
        }
        cur = cur[part];
    }

    return cur === undefined || cur === null ? fallback : cur;
}
