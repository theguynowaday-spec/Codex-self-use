ES5 Practice Script: MediaWiki/Fandom Safe Batch Editor (Dry-run + Apply)
=========================================================================

Description
-----------
This is a practice script in ES5 style for MediaWiki/Fandom workflows.
It demonstrates:
- reusable XHR wrappers
- GET/POST API helpers
- continuation pagination
- safe edit flow with token + conflict fields
- retry + backoff
- queue processing
- dry-run and apply modes

IMPORTANT
---------
- This is a practice template and should be reviewed before production use.
- Update BASE_API, page lists, and transform logic for your wiki.
- For browser cross-origin requests you may need `origin=*` for read requests.

(function () {
    "use strict";

    var CONFIG = {
        BASE_API: "https://example.fandom.com/api.php",
        USE_ORIGIN: true,
        USER_AGENT_NOTE: "ES5-Practice-Script",
        DRY_RUN: true,
        EDIT_DELAY_MS: 1500,
        MAX_RETRIES: 3,
        RETRY_BASE_DELAY_MS: 1000,
        SUMMARY: "ES5 bot: normalize spacing and add maintenance category",
        TARGET_CATEGORY: "Category:Example pages",
        MAX_PAGES: 30
    };

    function log() {
        if (typeof console !== "undefined" && console.log) {
            console.log.apply(console, arguments);
        }
    }

    function safeString(value) {
        return value === undefined || value === null ? "" : String(value);
    }

    function encodeForm(data) {
        var parts = [];
        var key;
        for (key in data) {
            if (data.hasOwnProperty(key) && data[key] !== undefined && data[key] !== null) {
                parts.push(encodeURIComponent(key) + "=" + encodeURIComponent(data[key]));
            }
        }
        return parts.join("&");
    }

    function buildUrl(base, params) {
        var q = encodeForm(params || {});
        if (!q) return base;
        return base + (base.indexOf("?") === -1 ? "?" : "&") + q;
    }

    function requestJson(options, onSuccess, onError) {
        var xhr = new XMLHttpRequest();
        var method = options.method || "GET";
        var url = options.url;
        var headers = options.headers || {};
        var body = options.body || null;
        var h;

        xhr.open(method, url, true);

        for (h in headers) {
            if (headers.hasOwnProperty(h)) {
                xhr.setRequestHeader(h, headers[h]);
            }
        }

        xhr.onreadystatechange = function () {
            var json;
            if (xhr.readyState !== 4) return;

            if (xhr.status >= 200 && xhr.status < 300) {
                try {
                    json = JSON.parse(xhr.responseText);
                    onSuccess(json, xhr);
                } catch (e) {
                    if (onError) onError(e, xhr);
                }
            } else {
                if (onError) onError(new Error("HTTP " + xhr.status), xhr);
            }
        };

        xhr.onerror = function () {
            if (onError) onError(new Error("Network error"), xhr);
        };

        xhr.send(body);
    }

    function retryAsync(taskFn, attempts, baseDelay, onSuccess, onError) {
        function run(left, attemptNum) {
            taskFn(function ok(result) {
                onSuccess(result);
            }, function fail(err) {
                var delay;
                if (left <= 1) {
                    onError(err);
                    return;
                }
                delay = baseDelay * Math.pow(2, attemptNum - 1);
                setTimeout(function () {
                    run(left - 1, attemptNum + 1);
                }, delay);
            });
        }
        run(attempts, 1);
    }

    function apiGet(params, onSuccess, onError) {
        var p = {};
        var key;

        for (key in params) {
            if (params.hasOwnProperty(key)) p[key] = params[key];
        }

        p.format = "json";
        if (CONFIG.USE_ORIGIN && !p.origin) p.origin = "*";

        requestJson({
            method: "GET",
            url: buildUrl(CONFIG.BASE_API, p),
            headers: {
                "X-Requested-With": "XMLHttpRequest"
            }
        }, onSuccess, onError);
    }

    function apiPost(params, onSuccess, onError) {
        var p = {};
        var key;

        for (key in params) {
            if (params.hasOwnProperty(key)) p[key] = params[key];
        }

        p.format = "json";

        requestJson({
            method: "POST",
            url: CONFIG.BASE_API,
            body: encodeForm(p),
            headers: {
                "Content-Type": "application/x-www-form-urlencoded; charset=UTF-8",
                "X-Requested-With": "XMLHttpRequest"
            }
        }, onSuccess, onError);
    }

    function getCsrfToken(onSuccess, onError) {
        retryAsync(function (ok, fail) {
            apiGet({
                action: "query",
                meta: "tokens",
                type: "csrf"
            }, function (data) {
                var token = data && data.query && data.query.tokens && data.query.tokens.csrftoken;
                if (!token) {
                    fail(new Error("CSRF token missing"));
                    return;
                }
                ok(token);
            }, fail);
        }, CONFIG.MAX_RETRIES, CONFIG.RETRY_BASE_DELAY_MS, onSuccess, onError);
    }

    function getCategoryMembers(categoryTitle, limit, onSuccess, onError) {
        var all = [];
        var cmcontinue = null;

        function step() {
            var params = {
                action: "query",
                list: "categorymembers",
                cmtitle: categoryTitle,
                cmlimit: Math.min(50, limit - all.length)
            };

            if (cmcontinue) params.cmcontinue = cmcontinue;

            apiGet(params, function (data) {
                var items = data && data.query && data.query.categorymembers ? data.query.categorymembers : [];
                var i;
                for (i = 0; i < items.length; i++) {
                    all.push(items[i]);
                    if (all.length >= limit) break;
                }

                if (all.length >= limit) {
                    onSuccess(all);
                    return;
                }

                if (data && data.continue && data.continue.cmcontinue) {
                    cmcontinue = data.continue.cmcontinue;
                    step();
                    return;
                }

                onSuccess(all);
            }, onError);
        }

        step();
    }

    function readPage(title, onSuccess, onError) {
        apiGet({
            action: "query",
            prop: "revisions",
            titles: title,
            rvprop: "content|timestamp|ids"
        }, function (data) {
            var pages = data && data.query && data.query.pages;
            var id;
            var page;
            var rev;
            var text;

            if (!pages) {
                onError(new Error("Missing query.pages"));
                return;
            }

            for (id in pages) {
                if (pages.hasOwnProperty(id)) {
                    page = pages[id];
                    if (page.missing) {
                        onError(new Error("Page missing: " + title));
                        return;
                    }

                    rev = page.revisions && page.revisions[0];
                    if (!rev) {
                        onError(new Error("No revision found: " + title));
                        return;
                    }

                    text = rev["*"] || (rev.slots && rev.slots.main && rev.slots.main["*"]) || "";
                    onSuccess({
                        title: page.title,
                        pageid: page.pageid,
                        revid: rev.revid,
                        basetimestamp: rev.timestamp,
                        text: text
                    });
                    return;
                }
            }

            onError(new Error("Page object not found: " + title));
        }, onError);
    }

    function normalizeWhitespace(text) {
        var out = safeString(text);
        out = out.replace(/[ \t]{2,}/g, " ");
        out = out.replace(/\n{3,}/g, "\n\n");
        out = out.replace(/[ \t]+\n/g, "\n");
        return out;
    }

    function addCategoryIfMissing(text, categoryName) {
        var marker = "[[" + categoryName + "]]";
        if (text.indexOf(marker) !== -1) return text;
        return text + "\n" + marker + "\n";
    }

    function transformPageText(text) {
        var out = normalizeWhitespace(text);
        out = addCategoryIfMissing(out, "Category:Maintenance");
        return out;
    }

    function needsChange(oldText, newText) {
        return safeString(oldText) !== safeString(newText);
    }

    function editPage(page, newText, token, onSuccess, onError) {
        apiPost({
            action: "edit",
            title: page.title,
            text: newText,
            summary: CONFIG.SUMMARY,
            token: token,
            basetimestamp: page.basetimestamp,
            starttimestamp: (new Date()).toISOString()
        }, function (data) {
            if (data && data.edit && data.edit.result === "Success") {
                onSuccess(data);
                return;
            }
            if (data && data.error) {
                onError(new Error(data.error.info || "API edit error"));
                return;
            }
            onError(new Error("Unknown edit response"));
        }, onError);
    }

    function runQueue(tasks, onDone) {
        var i = 0;

        function next() {
            if (i >= tasks.length) {
                if (onDone) onDone();
                return;
            }
            tasks[i++](next);
        }

        next();
    }

    function processPageTitle(title, token, done) {
        retryAsync(function (ok, fail) {
            readPage(title, function (page) {
                var updated = transformPageText(page.text);

                if (!needsChange(page.text, updated)) {
                    log("[SKIP]", title, "no changes");
                    ok({ title: title, status: "skip" });
                    return;
                }

                if (CONFIG.DRY_RUN) {
                    log("[DRY-RUN]", title, "would edit", "oldLen=", page.text.length, "newLen=", updated.length);
                    ok({ title: title, status: "dry-run" });
                    return;
                }

                setTimeout(function () {
                    editPage(page, updated, token, function () {
                        log("[EDITED]", title);
                        ok({ title: title, status: "edited" });
                    }, fail);
                }, CONFIG.EDIT_DELAY_MS);
            }, fail);
        }, CONFIG.MAX_RETRIES, CONFIG.RETRY_BASE_DELAY_MS, function () {
            done();
        }, function (err) {
            log("[ERROR]", title, err.message);
            done();
        });
    }

    function buildTasksFromMembers(members, token) {
        var tasks = [];
        var i;

        for (i = 0; i < members.length; i++) {
            (function (item) {
                tasks.push(function (next) {
                    if (item.ns !== 0) {
                        log("[SKIP]", item.title, "ns=", item.ns);
                        next();
                        return;
                    }
                    processPageTitle(item.title, token, next);
                });
            }(members[i]));
        }

        return tasks;
    }

    function main() {
        log("=== ES5 Practice Script Started ===");
        log("DRY_RUN:", CONFIG.DRY_RUN);
        log("BASE_API:", CONFIG.BASE_API);

        getCsrfToken(function (token) {
            log("Token acquired");

            getCategoryMembers(CONFIG.TARGET_CATEGORY, CONFIG.MAX_PAGES, function (members) {
                var tasks = buildTasksFromMembers(members, token);
                log("Members loaded:", members.length);
                log("Tasks queued:", tasks.length);

                runQueue(tasks, function () {
                    log("=== Script Finished ===");
                });
            }, function (err) {
                log("Category load failed:", err.message);
            });
        }, function (err) {
            log("Token load failed:", err.message);
        });
    }

    main();
}());

Notes for discussion
--------------------
1) To run as a true edit bot, set CONFIG.DRY_RUN = false.
2) For external scripts, keep session/cookies/token flow consistent.
3) For browser cross-wiki reads, origin=* may be required.
4) For production, add structured logs and persistent progress checkpointing.
5) Prefer reviewing diffs before apply mode.
